// Orbit Study — autonomous particle system
import { Body, Color } from "@loop/core";

export function setup(world) {
  world.set({
    gravity: 0.0,
    background: "#121315",
    seed: 42
  });

  for (let i = 0; i < 12; i++) {
    world.add(new Body({
      position: randomPoint(180),
      radius: random(5, 14),
      color: i % 4 === 0
        ? Color.chalk()
        : Color.signal(),
      glow: 0.8,
      trail: 36
    }));
  }
}

export function update(world, dt) {
  for (const body of world.bodies) {
    const pull = world.center
      .sub(body.position)
      .normalize()
      .scale(0.018);

    body.applyForce(pull);
    body.velocity.rotate(dt * 0.04);
  }
}
